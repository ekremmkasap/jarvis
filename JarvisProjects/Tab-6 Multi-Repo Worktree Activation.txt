from __future__ import annotations

import json
import logging
import os
import subprocess
import threading
from datetime import datetime, timezone
from pathlib import Path
from tempfile import NamedTemporaryFile
from typing import Any

import yaml

log = logging.getLogger("repo_manager")

ROOT = Path(__file__).resolve().parent.parent
CONFIG_PATH = ROOT / "config" / "repos.yaml"
STATUS_DIR = ROOT / "state" / "repos"
LOG_FILE = ROOT / "server" / "logs" / "repo_manager.log"

_lock = threading.RLock()


# ─────────────────── helpers ───────────────────

def _now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def _write_atomic(path: Path, data: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with NamedTemporaryFile("w", encoding="utf-8", delete=False,
                            dir=str(path.parent), suffix=".tmp") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
        tmp = Path(f.name)
    tmp.replace(path)


def _load_config() -> dict:
    if not CONFIG_PATH.exists():
        return {"repos": [], "settings": {}}
    try:
        with CONFIG_PATH.open(encoding="utf-8") as f:
            return yaml.safe_load(f) or {}
    except Exception as e:
        log.error(f"repos.yaml read error: {e}")
        return {"repos": [], "settings": {}}


def _status_path(repo_id: str) -> Path:
    return STATUS_DIR / f"{repo_id}.json"


def _load_status(repo_id: str) -> dict:
    path = _status_path(repo_id)
    if not path.exists():
        return {"id": repo_id, "cloned": False, "last_pull": None,
                "last_error": None, "branch": None, "commit": None}
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except Exception:
        return {"id": repo_id, "cloned": False, "last_pull": None,
                "last_error": None, "branch": None, "commit": None}


def _save_status(repo_id: str, data: dict) -> None:
    _write_atomic(_status_path(repo_id), data)


def _run_git(args: list[str], cwd: Path,
             timeout: int = 120) -> tuple[bool, str]:
    try:
        result = subprocess.run(
            ["git"] + args,
            cwd=str(cwd),
            capture_output=True,
            text=True,
            timeout=timeout,
        )
        if result.returncode == 0:
            return True, result.stdout.strip()
        return False, result.stderr.strip() or result.stdout.strip()
    except subprocess.TimeoutExpired:
        return False, f"git timeout ({timeout}s)"
    except FileNotFoundError:
        return False, "git bulunamadi. PATH kontrolu yap."
    except Exception as e:
        return False, str(e)


def _get_current_commit(repo_path: Path) -> str:
    ok, out = _run_git(["rev-parse", "--short", "HEAD"], repo_path)
    return out if ok else "unknown"


def _get_current_branch(repo_path: Path) -> str:
    ok, out = _run_git(["branch", "--show-current"], repo_path)
    return out if ok else "unknown"


# ─────────────────── public API ───────────────────

def list_repos() -> list[dict]:
    config = _load_config()
    repos = config.get("repos") or []
    result = []
    for repo in repos:
        if not isinstance(repo, dict):
            continue
        repo_id = str(repo.get("id") or "").strip()
        local_path = ROOT / str(repo.get("local_path") or f"repos/{repo_id}")
        status = _load_status(repo_id)
        cloned = local_path.exists() and (local_path / ".git").exists()
        result.append({
            "id": repo_id,
            "url": repo.get("url") or "",
            "branch": repo.get("branch") or "main",
            "local_path": str(local_path),
            "description": repo.get("description") or "",
            "tags": repo.get("tags") or [],
            "active": bool(repo.get("active")),
            "cloned": cloned,
            "last_pull": status.get("last_pull"),
            "last_error": status.get("last_error"),
            "current_branch": status.get("branch"),
            "current_commit": status.get("commit"),
        })
    return result


def get_repo(repo_id: str) -> dict | None:
    for repo in list_repos():
        if repo["id"] == repo_id:
            return repo
    return None


def clone_repo(repo_id: str) -> dict:
    config = _load_config()
    repos = config.get("repos") or []
    repo_conf = next((r for r in repos if r.get("id") == repo_id), None)

    if not repo_conf:
        return {"ok": False, "error": f"Repo bulunamadi: {repo_id}"}

    url = str(repo_conf.get("url") or "").strip()
    if not url:
        return {"ok": False, "error": f"{repo_id} icin URL tanimlanmamis."}

    local_path = ROOT / str(repo_conf.get("local_path") or f"repos/{repo_id}")

    if local_path.exists() and (local_path / ".git").exists():
        return {"ok": True, "message": f"{repo_id} zaten klonlanmis.", "path": str(local_path)}

    local_path.parent.mkdir(parents=True, exist_ok=True)
    branch = str(repo_conf.get("branch") or "main")
    log.info(f"Cloning {repo_id} from {url}...")

    ok, out = _run_git(
        ["clone", "--branch", branch, "--depth", "1", url, str(local_path)],
        cwd=ROOT,
        timeout=120,
    )

    status = _load_status(repo_id)
    if ok:
        status.update({
            "cloned": True,
            "last_pull": _now_iso(),
            "last_error": None,
            "branch": _get_current_branch(local_path),
            "commit": _get_current_commit(local_path),
        })
        _save_status(repo_id, status)
        log.info(f"Clone OK: {repo_id}")
        return {"ok": True, "message": f"{repo_id} basariyla klonlandi.", "path": str(local_path)}
    else:
        status["last_error"] = out
        _save_status(repo_id, status)
        log.error(f"Clone FAIL: {repo_id} — {out}")
        return {"ok": False, "error": out}


def pull_repo(repo_id: str) -> dict:
    repo = get_repo(repo_id)
    if not repo:
        return {"ok": False, "error": f"Repo bulunamadi: {repo_id}"}

    local_path = Path(repo["local_path"])
    if not local_path.exists() or not (local_path / ".git").exists():
        return {"ok": False, "error": f"{repo_id} klonlanmamis. Once clone et."}

    log.info(f"Pulling {repo_id}...")
    ok, out = _run_git(["pull", "--ff-only"], cwd=local_path, timeout=60)

    status = _load_status(repo_id)
    if ok:
        status.update({
            "last_pull": _now_iso(),
            "last_error": None,
            "branch": _get_current_branch(local_path),
            "commit": _get_current_commit(local_path),
        })
        _save_status(repo_id, status)
        return {"ok": True, "message": f"{repo_id} guncellendi.", "output": out[:200]}
    else:
        status["last_error"] = out
        _save_status(repo_id, status)
        return {"ok": False, "error": out[:200]}


def pull_all_active() -> list[dict]:
    results = []
    for repo in list_repos():
        if repo["active"] and repo["cloned"]:
            results.append({"id": repo["id"], **pull_repo(repo["id"])})
    return results


def get_repo_status_summary() -> dict:
    repos = list_repos()
    cloned = [r for r in repos if r["cloned"]]
    active = [r for r in repos if r["active"]]
    errors = [r for r in repos if r.get("last_error")]
    return {
        "total": len(repos),
        "cloned": len(cloned),
        "active": len(active),
        "errors": len(errors),
        "repos": repos,
    }


def activate_repo(repo_id: str) -> dict:
    """repos.yaml'da active: true yap."""
    try:
        config = _load_config()
        repos = config.get("repos") or []
        for repo in repos:
            if repo.get("id") == repo_id:
                repo["active"] = True
                with CONFIG_PATH.open("w", encoding="utf-8") as f:
                    yaml.dump(config, f, allow_unicode=True, default_flow_style=False)
                return {"ok": True, "message": f"{repo_id} aktive edildi."}
        return {"ok": False, "error": f"Repo bulunamadi: {repo_id}"}
    except Exception as e:
        return {"ok": False, "error": str(e)}


def deactivate_repo(repo_id: str) -> dict:
    try:
        config = _load_config()
        repos = config.get("repos") or []
        for repo in repos:
            if repo.get("id") == repo_id:
                repo["active"] = False
                with CONFIG_PATH.open("w", encoding="utf-8") as f:
                    yaml.dump(config, f, allow_unicode=True, default_flow_style=False)
                return {"ok": True, "message": f"{repo_id} deaktive edildi."}
        return {"ok": False, "error": f"Repo bulunamadi: {repo_id}"}
    except Exception as e:
        return {"ok": False, "error": str(e)}

'use client';
import { useEffect, useState } from 'react';

const BRIDGE = process.env.NEXT_PUBLIC_BRIDGE_API || 'http://127.0.0.1:8081';

interface RepoRecord {
  id: string;
  url: string;
  branch: string;
  local_path: string;
  description: string;
  tags: string[];
  active: boolean;
  cloned: boolean;
  last_pull: string | null;
  last_error: string | null;
  current_branch: string | null;
  current_commit: string | null;
}

function statusColor(repo: RepoRecord): string {
  if (repo.last_error) return 'border-red-500/40 bg-red-500/10 text-red-300';
  if (repo.active && repo.cloned) return 'border-emerald-500/40 bg-emerald-500/10 text-emerald-300';
  if (repo.cloned) return 'border-sky-500/40 bg-sky-500/10 text-sky-300';
  return 'border-neutral-700 bg-neutral-800/50 text-neutral-400';
}

function statusLabel(repo: RepoRecord): string {
  if (repo.last_error) return 'HATA';
  if (repo.active && repo.cloned) return 'AKTİF';
  if (repo.cloned) return 'KLONLU';
  return 'PASİF';
}

export default function ReposPage() {
  const [repos, setRepos] = useState<RepoRecord[]>([]);
  const [message, setMessage] = useState('');
  const [loading, setLoading] = useState(true);

  async function refresh() {
    const r = await fetch(`${BRIDGE}/api/repos`, { cache: 'no-store' });
    const data = await r.json();
    setRepos(data.repos || []);
    setLoading(false);
  }

  useEffect(() => {
    refresh();
    const t = setInterval(refresh, 10000);
    return () => clearInterval(t);
  }, []);

  async function action(endpoint: string, body: object) {

      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    });
    const data = await r.json();
    setMessage(data.message || data.error || JSON.stringify(data));
    await refresh();
  }

  return (
    <div className="min-h-screen bg-neutral-950 px-6 py-6 font-mono text-neutral-100">
      <div className="mx-auto max-w-6xl space-y-6">
        <header className="border-b border-neutral-800 pb-4">
          <a href="/" className="text-xs text-neutral-500 hover:text-neutral-300 uppercase tracking-widest">Dashboard</a>
          <h1 className="mt-2 text-2xl font-semibold">Repo Manager</h1>
          <p className="mt-1 text-sm text-neutral-400">Multi-repo klonlama, güncelleme ve durum takibi.</p>
          {message && <p className="mt-2 text-sm text-emerald-400">{message}</p>}
        </header>

        <div className="flex gap-3">
          <button onClick={() => action('/api/repos/pull-all', {})}
            className="border border-neutral-700 px-3 py-1 text-xs uppercase tracking-widest text-neutral-200 rounded hover:bg-neutral-800">
            Hepsini Güncelle
          </button>
        </div>

        {loading ? (
          <p className="text-neutral-500">Yükleniyor...</p>
        ) : (
          <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
            {repos.map(repo => (
              <div key={repo.id} className="border border-neutral-800 bg-neutral-900 p-4 rounded-md space-y-3">
                <div className="flex items-start justify-between">
                  <div>
                    <div className="text-sm font-semibold text-white">{repo.id}</div>
                    <div className="text-xs text-neutral-500 mt-1">{repo.description}</div>
                  </div>
                  <span className={`text-[10px] px-2 py-1 rounded border uppercase tracking-widest ${statusColor(repo)}`}>
                    {statusLabel(repo)}
                  </span>
                </div>

                <div className="text-xs text-neutral-400 space-y-1">
                  <div>branch: {repo.current_branch || repo.branch}</div>
                  <div>commit: {repo.current_commit || '-'}</div>
                  <div>son güncelleme: {repo.last_pull ? new Date(repo.last_pull).toLocaleString('tr-TR') : '-'}</div>
                  {repo.last_error && <div className="text-red-400">hata: {repo.last_error.slice(0, 80)}</div>}
                  <div className="flex flex-wrap gap-1 mt-1">
                    {(repo.tags || []).map(tag => (
                      <span key={tag} className="border border-neutral-700 px-1 text-[10px] text-neutral-500 rounded">{tag}</span>
                    ))}
                  </div>
                </div>

WORKTREE_STATUS_PATH = ROOT / "state" / "worktree_status.json"
SLOTS = ("atlas", "forge", "nexus", "shield", "spark")

def _load_wt_status() -> dict:
    if not WORKTREE_STATUS_PATH.exists():
        return {s: {"exists": False, "branch": None, "last_used": None,
                    "last_error": None, "healthy": False} for s in SLOTS}
    try:
        return json.loads(WORKTREE_STATUS_PATH.read_text(encoding="utf-8"))
    except Exception:
        return {}

def _save_wt_status(data: dict) -> None:
    WORKTREE_STATUS_PATH.parent.mkdir(parents=True, exist_ok=True)
    with NamedTemporaryFile("w", encoding="utf-8", delete=False,
                            dir=str(WORKTREE_STATUS_PATH.parent), suffix=".tmp") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
        Path(f.name).replace(WORKTREE_STATUS_PATH)

def get_worktree_status() -> dict:
    status = _load_wt_status()
    manager = WorkspaceManager()
    for slot in SLOTS:
        path = manager.get_path(slot)
        exists = path.exists() and (path / ".git").exists()
        entry = status.setdefault(slot, {})
        entry["exists"] = exists
        if exists:
            ok, branch = True, manager.get_branch(slot)
            entry["branch"] = branch
            entry["healthy"] = True
        else:
            entry["healthy"] = False
    _save_wt_status(status)
    return status

def activate_all_worktrees() -> dict:
    """Tüm 5 slot için worktree oluştur. Varsa atla."""
    results = {}
    manager = WorkspaceManager()
    status = _load_wt_status()
    for slot in SLOTS:
        try:
            path = manager.ensure(slot)
            status[slot] = {
                "exists": True,
                "branch": manager.get_branch(slot),
                "last_used": _now_iso(),
                "last_error": None,
                "healthy": True,
            }
            results[slot] = {"ok": True, "path": str(path)}
        except Exception as e:
            status[slot] = {
                "exists": False,
                "branch": None,
                "last_used": None,
                "last_error": str(e),
                "healthy": False,
            }
            results[slot] = {"ok": False, "error": str(e)}
    _save_wt_status(status)
    return results

def deactivate_worktree(slot: str) -> dict:
    """Worktree'yi kaldır (git worktree remove)."""
    manager = WorkspaceManager()
    path = manager.get_path(slot)
    if not path.exists():
        return {"ok": True, "message": f"{slot} zaten yok."}
    try:
        result = subprocess.run(
            ["git", "worktree", "remove", "--force", str(path)],
            cwd=manager.ROOT_DIR,
            capture_output=True, text=True, timeout=30
        )
        if result.returncode == 0:
            status = _load_wt_status()
            status[slot] = {"exists": False, "branch": None,
                            "last_used": None, "last_error": None, "healthy": False}
            _save_wt_status(status)
            return {"ok": True, "message": f"{slot} worktree kaldirildi."}
        return {"ok": False, "error": result.stderr.strip()}
    except Exception as e:
        return {"ok": False, "error": str(e)}

def worktree_health_check() -> dict:
    """Tüm worktree'lerin sağlık raporu."""
    status = get_worktree_status()
    healthy = sum(1 for s in status.values() if s.get("healthy"))
    return {
        "total": len(SLOTS),
        "healthy": healthy,
        "unhealthy": len(SLOTS) - healthy,
        "slots": status,
    }

import { useEffect, useState } from 'react';

const BRIDGE = process.env.NEXT_PUBLIC_BRIDGE_API || 'http://127.0.0.1:8081';
const SLOTS = ['atlas', 'forge', 'nexus', 'shield', 'spark'] as const;
type Slot = typeof SLOTS[number];

interface SlotStatus {
  exists: boolean;
  branch: string | null;
  last_used: string | null;
  last_error: string | null;
  healthy: boolean;
}

interface HealthPayload {
  total: number;
  healthy: number;
  unhealthy: number;
  slots: Record<Slot, SlotStatus>;
}

const SLOT_ROLES: Record<string, string> = {
  atlas: 'Manager/Core',
  forge: 'Backend Ops',
  nexus: 'Overflow/Reserve',
  shield: 'Security/Audit',
  spark: 'Voice/Video',
};

function slotColor(slot: string): string {
  const map: Record<string, string> = {
    atlas: '#00bfff', forge: '#00ff88', nexus: '#ffdd00',
    shield: '#9b59b6', spark: '#ff69b4',
  };
  return map[slot] || '#888';
}

export default function WorktreesPage() {
  const [health, setHealth] = useState<HealthPayload | null>(null);
  const [message, setMessage] = useState('');
  const [loading, setLoading] = useState(true);

  async function refresh() {
    const r = await fetch(`${BRIDGE}/api/worktrees/health`, { cache: 'no-store' });
    setHealth(await r.json());
    setLoading(false);
  }

  useEffect(() => {
    refresh();
    const t = setInterval(refresh, 5000);
    return () => clearInterval(t);
  }, []);

  async function action(endpoint: string, body: object) {
    const r = await fetch(`${BRIDGE}${endpoint}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    });
    const data = await r.json();
    setMessage(data.message || data.error || 'Tamamlandi.');
    await refresh();
  }

  const slots = health?.slots || {} as Record<string, SlotStatus>;

  return (
    <div className="min-h-screen bg-neutral-950 px-6 py-6 font-mono text-neutral-100">
      <div className="mx-auto max-w-5xl space-y-6">

        </header>
