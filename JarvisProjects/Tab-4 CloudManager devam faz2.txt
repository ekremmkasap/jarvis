  def get_instance_metrics(instance_id: str, hours: int = 1) -> dict:
      """
      CloudWatch'tan CPU, NetworkIn, NetworkOut metriklerini çeker.
      Returns: {
          ok: bool,
          instance_id: str,
          cpu_avg: float,
          network_in_mb: float,
          network_out_mb: float,
          period_hours: int
      }
      """
      try:
          import boto3, os
          from datetime import datetime, timedelta, timezone
          cw = boto3.client(
              "cloudwatch",
              aws_access_key_id=os.getenv("AWS_ACCESS_KEY_ID"),
              aws_secret_access_key=os.getenv("AWS_SECRET_ACCESS_KEY"),
              region_name=os.getenv("AWS_DEFAULT_REGION", "us-east-1"),
          )
          end = datetime.now(timezone.utc)
          start = end - timedelta(hours=hours)

          def _get_metric(metric_name, unit):
              resp = cw.get_metric_statistics(
                  Namespace="AWS/EC2",
                  MetricName=metric_name,
                  Dimensions=[{"Name": "InstanceId", "Value": instance_id}],
                  StartTime=start,
                  EndTime=end,
                  Period=3600,
                  Statistics=["Average"],
                  Unit=unit,
              )
              points = resp.get("Datapoints", [])
              if not points:
                  return 0.0
              return round(sum(p["Average"] for p in points) / len(points), 2)

          cpu = _get_metric("CPUUtilization", "Percent")
          net_in = round(_get_metric("NetworkIn", "Bytes") / (1024*1024), 3)
          net_out = round(_get_metric("NetworkOut", "Bytes") / (1024*1024), 3)

          return {
              "ok": True,
              "instance_id": instance_id,
              "cpu_avg": cpu,
              "network_in_mb": net_in,
              "network_out_mb": net_out,
              "period_hours": hours,
          }
      except Exception as e:
          return {"ok": False, "error": str(e)}

  def reboot_instance(instance_id: str) -> dict:
      """Instance'ı yeniden başlatır. Returns: {ok, message}"""
      try:
          import boto3, os
          ec2 = boto3.client(
              "ec2",
              aws_access_key_id=os.getenv("AWS_ACCESS_KEY_ID"),
              aws_secret_access_key=os.getenv("AWS_SECRET_ACCESS_KEY"),
              region_name=os.getenv("AWS_DEFAULT_REGION", "us-east-1"),
          )
          ec2.reboot_instances(InstanceIds=[instance_id])
          return {"ok": True, "message": f"{instance_id} yeniden baslatildi."}
      except Exception as e:
          return {"ok": False, "error": str(e)}

  def generate_presigned_url(bucket: str, key: str, expires_in: int = 3600) -> dict:
      """
      S3 objesi için geçici indirme URL'i üretir.
      Returns: {ok, url, expires_in, bucket, key}
      """
      try:
          import boto3, os
          s3 = boto3.client(
              "s3",
              aws_access_key_id=os.getenv("AWS_ACCESS_KEY_ID"),
              aws_secret_access_key=os.getenv("AWS_SECRET_ACCESS_KEY"),
              region_name=os.getenv("AWS_DEFAULT_REGION", "us-east-1"),
          )
          url = s3.generate_presigned_url(
              "get_object",
              Params={"Bucket": bucket, "Key": key},
              ExpiresIn=expires_in,
          )
          return {"ok": True, "url": url, "expires_in": expires_in, "bucket": bucket, "key": key}
      except Exception as e:
          return {"ok": False, "error": str(e)}

  def get_object_metadata(bucket: str, key: str) -> dict:
      """S3 objesinin metadata'sını döner. Returns: {ok, content_type, size_bytes, last_modified}"""
      try:
          import boto3, os
          s3 = boto3.client(
              "s3",
              aws_access_key_id=os.getenv("AWS_ACCESS_KEY_ID"),
              aws_secret_access_key=os.getenv("AWS_SECRET_ACCESS_KEY"),
              region_name=os.getenv("AWS_DEFAULT_REGION", "us-east-1"),
          )
          resp = s3.head_object(Bucket=bucket, Key=key)
          return {
              "ok": True,
              "content_type": resp.get("ContentType", "unknown"),
              "size_bytes": resp.get("ContentLength", 0),
              "last_modified": str(resp.get("LastModified", "")),
          }
      except Exception as e:
          return {"ok": False, "error": str(e)}

  def check_cost_alerts() -> list[dict]:
      """
      state/cost_alerts.json'daki threshold'ları kontrol eder.
      Returns: [{service, limit_usd, current_usd, pct_used, alert: bool}]
      """
      import json
      from pathlib import Path
      alerts_path = Path(__file__).parent.parent.parent / "state" / "cost_alerts.json"
      if not alerts_path.exists():
          return []
      try:
          alerts = json.loads(alerts_path.read_text(encoding="utf-8"))
      except Exception:
          return []
      cost_data = get_monthly_cost()
      by_service = cost_data.get("by_service", {})
      results = []
      for alert in (alerts if isinstance(alerts, list) else []):
          service = alert.get("service", "")
          limit = float(alert.get("usd_limit", 0))
          current = float(by_service.get(service, 0))
          pct = round((current / limit * 100), 1) if limit > 0 else 0
          results.append({
              "service": service,
              "limit_usd": limit,
              "current_usd": current,
              "pct_used": pct,
              "alert": pct >= 80,
          })
      return results

  def get_cost_summary_text() -> str:
      """Telegram için kısa maliyet özeti. Max 300 karakter."""
      data = get_monthly_cost()
      total = data.get("total_usd", 0)
      by_service = data.get("by_service", {})
      top = sorted(by_service.items(), key=lambda x: x[1], reverse=True)[:3]
      lines = [f"Bu ay: ${total:.2f}"]
      for svc, cost in top:
          lines.append(f"  {svc}: ${cost:.2f}")
      alerts = check_cost_alerts()
      triggered = [a for a in alerts if a.get("alert")]
      if triggered:
          lines.append(f"⚠ {len(triggered)} uyari esigi asildi!")
      return "\n".join(lines)[:300]
