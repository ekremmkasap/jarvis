@echo off

)

:menu
cls
echo.
echo  ================================================
echo   JARVIS MARK-NEXUS v2.0
echo   8 Agent AI Operating System
echo  ================================================
echo.
echo   [1] Normal Jarvis     (Mark-XXXV + Hologram)
echo   [2] Swarm Modu        (CEO + 7 Ajan Sesli)

echo   [0] Cikis

if "!SECIM!"=="1" goto normal
if "!SECIM!"=="2" goto swarm
if "!SECIM!"=="3" goto webonly
if "!SECIM!"=="4" goto silent
if "!SECIM!"=="5" goto stop
if "!SECIM!"=="0" exit /b 0
echo Gecersiz secim, tekrar deneyin.
timeout /t 2 >nul
goto menu

:normal
cls
echo.
echo  Jarvis Normal Mod baslatiliyor...

echo  Repo  : %REPO%
echo.

    pause
    goto menu
)
set "JARVIS_ENABLE_HOLOGRAM=1"
set "JARVIS_SWARM_HOLOGRAM=0"
set "JARVIS_VOICE_RUNTIME=mark_xxxv"
set "PYTHONIOENCODING=utf-8"

echo  CEO baslatiliyor: !SWARM_KONU!
echo.
set "JARVIS_ENABLE_HOLOGRAM=1"

set "JARVIS_VOICE_RUNTIME=mark_xxxv"
