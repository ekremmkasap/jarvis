1. web_search executor:
   def _web_search_executor(payload, context):
       query = payload.get("query", "")
       try:
           # Mevcut web search skill'i kullan (varsa)
           # Yoksa basit requests ile DuckDuckGo Instant Answer API
           import requests
           resp = requests.get(
               "https://api.duckduckgo.com/",
               params={"q": query, "format": "json", "no_html": 1},
               timeout=10
           )
           data = resp.json()
           answer = data.get("AbstractText") or data.get("Answer") or "Sonuç bulunamadı."
           return {"ok": True, "result": answer, "query": query}
       except Exception as e:
           return {"ok": False, "error": str(e), "query": query}

2. file_reader executor:
   def _file_reader_executor(payload, context):
       filepath = payload.get("path", "")
       try:
           from pathlib import Path
           content = Path(filepath).read_text(encoding="utf-8", errors="ignore")
           return {"ok": True, "result": content[:3000], "path": filepath}  # max 3000 char
       except Exception as e:
           return {"ok": False, "error": str(e), "path": filepath}

3. code_analyzer executor:
   def _code_analyzer_executor(payload, context):
       code = payload.get("code", "") or payload.get("content", "")
       filepath = payload.get("path", "")
       if filepath and not code:
           try:
               from pathlib import Path
               code = Path(filepath).read_text(encoding="utf-8", errors="ignore")[:5000]
           except Exception:
               pass
       # Basit statik analiz: satır sayısı, fonksiyon sayısı, TODO sayısı
       lines = code.splitlines()
       result = {
           "lines": len(lines),
           "functions": sum(1 for l in lines if l.strip().startswith("def ")),
           "todos": sum(1 for l in lines if "TODO" in l or "FIXME" in l),
           "preview": "\n".join(lines[:20]),
       }
       return {"ok": True, "result": result}

4. obsidian_writer executor:
   def _obsidian_writer_executor(payload, context):
       try:
           from server.skills.persona_obsidian_skill import save_note
           persona_id = context.get("persona_id", "jarvis")
           title = payload.get("title", "Swarm Notu")
           content = payload.get("content", "")
           note = save_note(persona_id=persona_id, title=title, content=content)
           return {"ok": True, "result": f"Kaydedildi: {note.get('path', '?')}"}
       except Exception as e:
           return {"ok": False, "error": str(e)}

5. summarizer executor:
   def _summarizer_executor(payload, context):
       text = payload.get("text", "") or payload.get("content", "")
       max_len = payload.get("max_length", 300)
       if not text:
           return {"ok": False, "error": "text_required"}
       # Basit summarizer: ilk N karakter + cümle sınırı
       summary = text[:max_len]
       last_period = summary.rfind(".")
       if last_period > 50:
           summary = summary[:last_period+1]
       return {"ok": True, "result": summary}

       """3+ adım gerektiren görevleri tespit et"""
       keywords = ["analiz et ve", "araştır ve", "bul ve", "incele ve", "yap ve raporla"]
       return any(kw in task_text.lower() for kw in keywords)

   def run_persona_swarm(persona_id: str, task: str, executor_registry=None) -> dict:
       """Bridge'den çağrılacak ana entry point"""
       if executor_registry is None:
           executor_registry = default_executor_registry()
       # Görevi adımlara böl
       steps = [
           {"type": "web_search", "payload": {"query": task}},
           {"type": "summarizer", "payload": {"text": "", "max_length": 500}},
           {"type": "obsidian_writer", "payload": {"title": task[:50], "content": ""}},
       ]
       # Persona context
       context = {"persona_id": persona_id, "task": task}
       return execute_steps(steps, executor_registry, context)
